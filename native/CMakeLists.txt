# CMake / native/CMakeLists.txt / CMake 3.20+ / Windows x64
cmake_minimum_required(VERSION 3.20)
project(AeonNative LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Rook requires Windows x64")
endif()
add_executable(Aeon WIN32 rook.cpp)
target_compile_features(Aeon PRIVATE cxx_std_20)
target_link_libraries(Aeon PRIVATE user32 gdi32 shell32 advapi32 bcrypt comctl32)
if(MSVC)
  target_compile_options(Aeon PRIVATE /W4 /EHsc /utf-8)
else()
  target_link_options(Aeon PRIVATE -municode)
endif()
